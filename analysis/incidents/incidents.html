def create_heatmap(df, order_mapping, col, start=None, end=None, lower_bound=None, upper_bound=None):
    order_mapping = {v: k for k, v in enumerate(order_mapping)}
    df['order'] = df['segment'].map(order_mapping)
    df.sort_values(by="order", inplace=True)

    heatmap_data = df.pivot(index='order', columns='measurement_time', values=col)

    # Plot de heatmap
    fig = px.imshow(
        heatmap_data,
        labels=dict(x="Measurement Time", y="Segment", color=col),
        aspect="auto",  
        color_continuous_scale='RdBu_r',  
        zmin=lower_bound,
        zmax=upper_bound
    )
    fig.update_xaxes(tickangle=45)  # draai tijdlabels iets voor leesbaarheid

    if start is not None:
        start = pd.to_datetime(start)
        fig.add_vline(
            x=start,
            line=dict(color="orange", width=2, dash="dot"),
           
        )
        fig.add_annotation(
            x=start,
            y=1.05,  # boven de plot met yref='paper'
            xref='x',
            yref='paper',
            text="Start",
            showarrow=False,
            font=dict(color="orange", size=12),
            align="center",
            textangle=0
        )
    if end is not None:
        end = pd.to_datetime(end)
        fig.add_vline(
            x=end,
            line=dict(color="orange", width=2, dash="dot"),
        )
        fig.add_annotation(
            x=end,
            y=1.05,  # boven de plot met yref='paper'
            xref='x',
            yref='paper',
            text="End",
            showarrow=False,
            font=dict(color="orange", size=12),
            align="center",
            textangle=0
        )

    
    return fig

def calculate_statistical_measures(df, cols=['speed_kph', '0_5']):
    for col in cols:
        df[f'Z_{col}'] = (df[col] - df[f'{col}_mean']) / df[f'{col}_std']
                
    return df
